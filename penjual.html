function tambahKartuPesanan(pesanan) { const container = document.getElementById('daftar-pesanan'); const isDone = pesanan.status === 'siap_ambil'; const isUnpaid = pesanan.status === 'menunggu_pembayaran'; // Rincian items pesanan let daftarItemsHtml = ''; const items = pesanan.items || {}; // Loop semua item yang tersimpan di Supabase for (let key in items) { const qty = parseInt(items[key]); if (qty > 0) { let namaBersih = key.replace('qty-', '').toUpperCase(); // merubah 'qty-tahu' jadi 'TAHU' if (key === 'qty-tahu') namaBersih = '🥟 Tahu Isi Crispy'; if (key === 'qty-tempe') namaBersih = '🟨 Tempe Mendoan'; if (key === 'qty-bakwan') namaBersih = '🥦 Bakwan Sayur'; daftarItemsHtml += `
  • ${namaBersih} x${qty}
    • `; } } const cardHtml = `
    #A-0${pesanan.id} ${isUnpaid ? '⏳ Menunggu Bayar' : (isDone ? '✅ Selesai' : '🔥 Siap Digoreng')}

    Tingkat Crispy: ${pesanan.crispy_level || 'Standard'}

    Item Pesanan:

      ${daftarItemsHtml || '
    • Tidak ada rincian item
      • '}
    Total Pembayaran: Rp ${(pesanan.total || 0).toLocaleString('id-ID')}
    ${isDone ? `✓ Gorengan Siap Diambil` : `` }
    `; container.insertAdjacentHTML('beforeend', cardHtml); }
